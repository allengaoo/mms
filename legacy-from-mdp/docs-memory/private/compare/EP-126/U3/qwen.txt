def cmd_template(args: argparse.Namespace) -> int:
    """template 子命令：代码模板库"""
    if args.action == 'create':
        template_path = os.path.join(args.output, f"{args.name}.py")
        with open(template_path, 'w') as f:
            if args.type == 'api':
                f.write(f"# {args.name} API Template\n\n"
                        "from fastapi import APIRouter, Depends, Query\n"
                        "from pydantic import BaseModel\n"
                        "\n"
                        "router = APIRouter()\n"
                        "\n"
                        "class {args.name}Response(BaseModel):\n"
                        "    id: int\n"
                        "    name: str\n"
                        "\n"
                        "@router.get(\"/{args.name.lower()}s\", response_model=List[{args.name}Response])\n"
                        "async def list_{args.name.lower()}():\n"
                        "    # Business logic should be in service layer\n"
                        "    return []\n"
                        "\n"
                        "if __name__ == \"__main__\":\n"
                        "    import uvicorn\n"
                        "    uvicorn.run(router, host=\"0.0.0.0\", port=8000)")
            elif args.type == 'service':
                f.write(f"# {args.name} Service Template\n\n"
                        "from abc import ABC, abstractmethod\n"
                        "from typing import List\n"
                        "\n"
                        "class {args.name}Service(ABC):\n"
                        "    @abstractmethod\n"
                        "    async def get_all(self) -> List[Dict[str, Any]]:\n"
                        "        pass\n"
                        "\n"
                        "class {args.name}ServiceImpl({args.name}Service):\n"
                        "    async def get_all(self) -> List[Dict[str, Any]]:\n"
                        "        # Implementation goes here\n"
                        "        return []")
            elif args.type == 'model':
                f.write(f"# {args.name} Model Template\n\n"
                        "from pydantic import BaseModel\n"
                        "\n"
                        "class {args.name}Model(BaseModel):\n"
                        "    id: int\n"
                        "    name: str\n"
                        "    description: str")

        ok(f"Created template '{args.name}' of type '{args.type}' at '{template_path}'")
    elif args.action == 'list':
        try:
            templates_dir = './templates'
            files = [f for f in os.listdir(templates_dir) if os.path.isfile(os.path.join(templates_dir, f))]
            if not files:
                info("No templates found.")
            else:
                info("Available templates:")
                for file in files:
                    print(f"- {file}")
        except FileNotFoundError:
            info("Templates directory not found.")
    return 0

def main() -> int:
    parser = build_parser()
    args = parser.parse_args()
    if not hasattr(args, 'command'):
        parser.print_help()
        return 1
    return cmd_template(args)
